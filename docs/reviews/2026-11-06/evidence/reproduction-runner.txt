import json,sys,os,ipaddress
from pathlib import Path
base=Path(os.environ['JARVIS_DATA_DIR']).resolve()
def guard(event,args):
    if event=='sqlite3.connect' and not Path(args[0]).resolve().is_relative_to(base): raise RuntimeError('DB outside reproduction forbidden')
    if event in ('socket.connect','socket.getaddrinfo'):
        host=args[1][0] if event=='socket.connect' else args[0]
        try: local=ipaddress.ip_address(host).is_loopback
        except ValueError: local=host in {'localhost',None}
        if not local: raise RuntimeError('External network forbidden')
    if event=='open' and isinstance(args[0],str) and (Path(args[0]).name=='.env' or Path(args[0]).absolute().is_relative_to(Path(r'C:\Users\matte\jarvis\data'))): raise RuntimeError('Protected input forbidden')
sys.addaudithook(guard)
